\begin{figure}[h]
    \centering
    \begin{tikzpicture}[very thick, line cap = round]
        \def\Lx{5}
        \def\Ly{6}
        \def\h{3*\Ly/10}
        \def\a{\Lx*4}
        \def\t{0.2}

        \fill[mydarkred!5] (0,0) rectangle (\Lx,\Ly);

        \draw[myblue, fill=mylightblue2, samples=20,variable=\y, domain=0.05:0.95] (0,0) -- plot({\a/(1/\y + 2/(1-\y))},{\y*\h}) -- (0,\h) -- cycle;
        \draw[myblue, fill=mylightblue2, samples=20,variable=\y, domain=1.05:1.95] (0,\h) -- plot({\a/(2/(\y-1) + 3/(2-\y))},{\y*\h}) -- (0,2*\h) -- cycle;
        \draw[myblue, fill=mylightblue2, samples=20,variable=\y, domain=2.05:2.95] (0,2*\h) -- plot({\a/(3/(\y-2) + 4/(3-\y))},{\y*\h}) -- (0,3*\h) -- cycle;
        
        \draw[->] (0,0) -- (\Lx,0) node[right] {$t/U$};
        \draw[->] (0,0) -- (0,\Ly) node[right] {$\mu/U$};

        \draw (\t/2,\h) --++ (-\t,0) node[left] {$1$};
        \draw (\t/2,2*\h) --++ (-\t,0) node[left] {$2$};
        \draw (\t/2,3*\h) --++ (-\t,0) node[left] {$3$};

        \node[mydarkblue] at (0.35*\Lx,0.45*\h) {MI ($1$)};
        \node[mydarkblue] at (0.2*\Lx,1.45*\h) {MI ($2$)};
        \node[mydarkblue] at (0.14*\Lx,2.5*\h) {MI ($3$)};
        \node[mydarkred] at (0.6*\Lx,2*\h) {Superfluid};
    \end{tikzpicture}
    \caption{Ground state phase diagram for the standard Bose-Hubbard model, showing the Mott-insulator to superfluid phase transition.}
    \label{figure-ground-state-phase-diagram-of-the-bose--hubbard-model}
\end{figure}
